<?php
$servername = "localhost";
$username = "root";
$password = "";
$dbname = "emlak_crm";

// Bağlantıyı oluştur
$conn = new mysqli($servername, $username, $password, $dbname);

// Bağlantıyı kontrol et
if ($conn->connect_error) {
    die("Connection failed: " . $conn->connect_error);
}

// Veritabanı yoksa oluştur
$conn->query("CREATE DATABASE IF NOT EXISTS $dbname");
$conn->select_db($dbname);

// Users tablosu
$conn->query("CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100) UNIQUE,
    password VARCHAR(255),
    role ENUM('admin','agent') DEFAULT 'agent'
)");

// Clients tablosu
$conn->query("CREATE TABLE IF NOT EXISTS clients (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    phone VARCHAR(20),
    email VARCHAR(100),
    request_type ENUM('satılık','kiralık'),
    rooms INT,
    price_range VARCHAR(50),
    floor INT,
    region VARCHAR(100),
    notes TEXT,
    status ENUM('yeni','görüşmede','tamamlandı') DEFAULT 'yeni',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)");

// Properties tablosu
$conn->query("CREATE TABLE IF NOT EXISTS properties (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(255),
    type ENUM('satılık','kiralık'),
    rooms INT,
    floor INT,
    area INT,
    age INT,
    deposit BOOLEAN DEFAULT FALSE,
    balcony BOOLEAN DEFAULT FALSE,
    heating VARCHAR(50),
    city_view BOOLEAN DEFAULT FALSE,
    region VARCHAR(100),
    price DECIMAL(15,2),
    image VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)");

// Admin hesabı ekle (sadece yoksa)
$adminEmail = "admin@emlakcrm.com";
$adminPass = password_hash("123456", PASSWORD_DEFAULT);
$result = $conn->query("SELECT * FROM users WHERE email='$adminEmail'");
if($result->num_rows == 0){
    $conn->query("INSERT INTO users (name,email,password,role) VALUES ('Admin','$adminEmail','$adminPass','admin')");
}
?>
